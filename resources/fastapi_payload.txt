DEFAULT_CONFIG = "alas"
CONFIG_DIR = "config"
LOG_DIR = "log"
LOG_TAIL_CHUNK_SIZE = 64 * 1024
DEFAULT_LOG_LINES = 200
MAX_LOG_LINES = 2000
MAX_CONFIG_UPLOAD_SIZE = 2 * 1024 * 1024
ALAS_GYRE_API_TOKEN = "__ALAS_GYRE_API_TOKEN__"
TOKEN_PLACEHOLDER = "__" + "ALAS_GYRE_API_TOKEN" + "__"
TOKEN_HEADER = "X-Alas-Gyre-Token"

STATE_MAP = {
    1: "running",
    2: "idle",
    3: "error",
    4: "update",
}
VALID_STATUSES = {"idle", "running", "error", "update", "disconnected"}
INVALID_CONFIG_NAME_CHARS = set('/\\:*?"<>|')
REQUIRED_ALAS_CONFIG_PATHS = {
    "Alas": ("Emulator", "Error", "Optimization", "Storage"),
    "General": ("Storage",),
    "Restart": ("Scheduler", "Storage"),
    "Main": ("Scheduler", "Campaign", "Fleet", "Storage"),
    "Reward": ("Scheduler", "Reward", "Storage"),
}
REQUIRED_ALAS_EMULATOR_KEYS = (
    "Serial",
    "PackageName",
    "ScreenshotMethod",
    "ControlMethod",
)
REQUIRED_ALAS_SCHEDULER_KEYS = ("Enable", "NextRun", "Command")
MIN_ALAS_CONFIG_SECTION_COUNT = 20
MIN_ALAS_CONFIG_SCHEDULER_COUNT = 5


def get_alas_root():
    return os.path.abspath(os.path.join(os.path.dirname(__file__), "..", ".."))


def get_data_dir(dirname):
    candidates = [
        os.path.abspath(dirname),
        os.path.join(get_alas_root(), dirname),
    ]
    for path in candidates:
        if os.path.isdir(path):
            return path
    return candidates[0]


def normalize_status(status):
    if isinstance(status, str) and status in VALID_STATUSES:
        return status
    return STATE_MAP.get(status, "idle")


def is_api_authorized(request):
    expected_token = ALAS_GYRE_API_TOKEN.strip()
    if not expected_token or expected_token == TOKEN_PLACEHOLDER:
        return False

    provided_token = request.headers.get(TOKEN_HEADER, "")
    return secrets.compare_digest(provided_token, expected_token)


def get_config_names():
    configs = []
    config_dir = get_data_dir(CONFIG_DIR)
    if os.path.isdir(config_dir):
        for entry in os.scandir(config_dir):
            if not entry.is_file():
                continue
            file_name = entry.name
            if not file_name.endswith(".json") or file_name.startswith("template"):
                continue
            configs.append(file_name[:-5])
    configs.sort(key=str.lower)
    return configs or [DEFAULT_CONFIG]


def get_default_config():
    configs = get_config_names()
    if DEFAULT_CONFIG in configs:
        return DEFAULT_CONFIG
    return configs[0] if configs else DEFAULT_CONFIG


def get_requested_config(request):
    config_name = request.query_params.get("config")
    if config_name is None:
        return get_default_config()
    return config_name.strip() or get_default_config()


def validate_requested_config(request, config_name):
    configs = get_config_names()
    if config_name not in configs:
        return JSONResponse(
            {
                "error": "unknown_config",
                "config": config_name,
                "configs": configs,
            },
            status_code=404,
        )
    return None


def normalize_config_name(config_name):
    name = str(config_name or "").strip()
    if name.endswith(".json"):
        name = name[:-5]
    if (
        not name
        or name in {".", ".."}
        or os.path.basename(name) != name
        or any(char in INVALID_CONFIG_NAME_CHARS for char in name)
        or any(ord(char) < 32 for char in name)
        or name.startswith("template")
        or len(name) > 120
    ):
        raise ValueError("invalid_config_name")
    return name


def raise_invalid_alas_config(reason, detail=""):
    if detail:
        raise ValueError(f"invalid_alas_config:{reason}:{detail}")
    raise ValueError(f"invalid_alas_config:{reason}")


def require_alas_config_dict(value, reason, detail):
    if not isinstance(value, dict):
        raise_invalid_alas_config(reason, detail)
    return value


def validate_alas_config_data(config_data):
    root = require_alas_config_dict(config_data, "root_not_object", "root")

    if len(root) < MIN_ALAS_CONFIG_SECTION_COUNT:
        raise_invalid_alas_config("too_few_sections", str(len(root)))

    missing_sections = [
        section for section in REQUIRED_ALAS_CONFIG_PATHS if section not in root
    ]
    if missing_sections:
        raise_invalid_alas_config("missing_section", ", ".join(missing_sections))

    for section, required_children in REQUIRED_ALAS_CONFIG_PATHS.items():
        section_data = require_alas_config_dict(
            root.get(section), "invalid_section", section
        )
        missing_children = [
            child for child in required_children if child not in section_data
        ]
        if missing_children:
            raise_invalid_alas_config(
                "missing_group",
                f"{section}.{', '.join(missing_children)}",
            )
        for child in required_children:
            require_alas_config_dict(
                section_data.get(child), "invalid_group", f"{section}.{child}"
            )

    emulator = root["Alas"]["Emulator"]
    missing_emulator_keys = [
        key for key in REQUIRED_ALAS_EMULATOR_KEYS if key not in emulator
    ]
    if missing_emulator_keys:
        raise_invalid_alas_config(
            "missing_emulator_key", ", ".join(missing_emulator_keys)
        )

    storage_errors = []
    for section in REQUIRED_ALAS_CONFIG_PATHS:
        storage = root[section].get("Storage")
        if not isinstance(storage, dict) or not isinstance(storage.get("Storage"), dict):
            storage_errors.append(section)
    if storage_errors:
        raise_invalid_alas_config("invalid_storage", ", ".join(storage_errors))

    scheduler_count = 0
    scheduler_errors = []
    for section, section_data in root.items():
        if not isinstance(section_data, dict) or "Scheduler" not in section_data:
            continue
        scheduler = section_data.get("Scheduler")
        if not isinstance(scheduler, dict):
            scheduler_errors.append(section)
            continue
        scheduler_count += 1
        missing_scheduler_keys = [
            key for key in REQUIRED_ALAS_SCHEDULER_KEYS if key not in scheduler
        ]
        if missing_scheduler_keys:
            scheduler_errors.append(f"{section}.{', '.join(missing_scheduler_keys)}")
            continue
        if not isinstance(scheduler.get("Enable"), bool):
            scheduler_errors.append(f"{section}.Enable")
        if not isinstance(scheduler.get("NextRun"), str):
            scheduler_errors.append(f"{section}.NextRun")
        if not isinstance(scheduler.get("Command"), str) or not scheduler.get("Command"):
            scheduler_errors.append(f"{section}.Command")

    if scheduler_count < MIN_ALAS_CONFIG_SCHEDULER_COUNT:
        raise_invalid_alas_config("too_few_schedulers", str(scheduler_count))
    if scheduler_errors:
        raise_invalid_alas_config(
            "invalid_scheduler", ", ".join(scheduler_errors[:5])
        )

    return True


def get_config_path(config_name):
    config_name = normalize_config_name(config_name)
    config_dir = os.path.abspath(get_data_dir(CONFIG_DIR))
    config_path = os.path.abspath(os.path.join(config_dir, f"{config_name}.json"))
    if os.path.commonpath([config_dir, config_path]) != config_dir:
        raise ValueError("invalid config path")
    return config_path


def get_manager(config_name):
    return ProcessManager.get_manager(config_name)


def get_status(config_name):
    manager = get_manager(config_name)
    return normalize_status(getattr(manager, "state", None))


def get_log_line_limit(request):
    try:
        value = int(request.query_params.get("lines", str(DEFAULT_LOG_LINES)))
    except (TypeError, ValueError):
        value = DEFAULT_LOG_LINES
    return max(1, min(value, MAX_LOG_LINES))


def get_latest_log_file(config_name):
    log_dir = get_data_dir(LOG_DIR)
    if not os.path.isdir(log_dir):
        return None

    suffix = f"_{config_name}.txt"
    candidates = []
    for entry in os.scandir(log_dir):
        if not entry.is_file() or not entry.name.endswith(suffix):
            continue
        candidates.append(entry.path)

    if not candidates:
        return None

    candidates.sort(
        key=lambda path: (os.path.getmtime(path), os.path.basename(path)),
        reverse=True,
    )
    return candidates[0]


def tail_log_file(log_file, line_limit):
    file_size = os.path.getsize(log_file)
    if file_size <= 0:
        return ""

    chunks = []
    newline_count = 0
    position = file_size

    with open(log_file, "rb") as f:
        while position > 0 and newline_count <= line_limit:
            read_size = min(LOG_TAIL_CHUNK_SIZE, position)
            position -= read_size
            f.seek(position)
            chunk = f.read(read_size)
            chunks.append(chunk)
            newline_count += chunk.count(b"\n")

    data = b"".join(reversed(chunks))
    lines = data.splitlines(keepends=True)
    return b"".join(lines[-line_limit:]).decode("utf-8", errors="replace")


def render_log_item(renderable):
    if isinstance(renderable, str):
        return renderable if renderable.endswith("\n") else renderable + "\n"
    if Console is None:
        return str(renderable) + "\n"

    console = Console(no_color=True, highlight=False, width=119)
    with console.capture() as capture:
        console.print(renderable)
    return capture.get()


def get_live_log(config_name, line_limit):
    manager = get_manager(config_name)
    renderables = getattr(manager, "renderables", None)
    if not renderables:
        return ""

    lines = deque(maxlen=line_limit)
    for renderable in renderables:
        lines.extend(render_log_item(renderable).splitlines(True))
    return "".join(lines)


class HeaderMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request, call_next):
        if (
            request.url.path.startswith("/api/")
            and request.method.upper() != "OPTIONS"
            and not is_api_authorized(request)
        ):
            response = JSONResponse({"error": "unauthorized"}, status_code=401)
        else:
            response = await call_next(request)
        response.headers["Cache-Control"] = "no-cache"
        response.headers["Access-Control-Allow-Origin"] = "*"
        response.headers["Access-Control-Allow-Methods"] = "GET, POST, DELETE, OPTIONS"
        response.headers["Access-Control-Allow-Headers"] = (
            "Content-Type, Authorization, X-Alas-Gyre-Token"
        )
        return response


def asgi_app(
    applications,
    cdn=True,
    static_dir=None,
    debug=False,
    allowed_origins=None,
    check_origin=None,
    **starlette_settings
):
    debug = Session.debug = os.environ.get("PYWEBIO_DEBUG", debug)
    cdn = cdn_validation(cdn, "warn")
    if cdn is False:
        cdn = "pywebio_static"
    routes = webio_routes(
        applications,
        cdn=cdn,
        allowed_origins=allowed_origins,
        check_origin=check_origin,
    )

    async def api_options(request):
        return JSONResponse({"ok": True})

    async def api_health(request):
        configs = get_config_names()
        return JSONResponse(
            {
                "ok": True,
                "configs": configs,
                "default": get_default_config(),
            }
        )

    async def api_get_configs(request):
        configs = get_config_names()
        return JSONResponse(
            {
                "configs": configs,
                "default": get_default_config(),
            }
        )

    async def api_upload_config(request):
        try:
            body = await request.body()
            if len(body) > MAX_CONFIG_UPLOAD_SIZE:
                return JSONResponse({"error": "config_too_large"}, status_code=413)

            payload = json.loads(body.decode("utf-8"))
            config_name = normalize_config_name(
                payload.get("name") or payload.get("config")
            )
            overwrite = bool(payload.get("overwrite", False))

            content = payload.get("content")
            if isinstance(content, str):
                if len(content.encode("utf-8")) > MAX_CONFIG_UPLOAD_SIZE:
                    return JSONResponse({"error": "config_too_large"}, status_code=413)
                config_data = json.loads(content)
            elif isinstance(content, dict):
                config_data = content
            else:
                return JSONResponse({"error": "invalid_config_content"}, status_code=400)

            validate_alas_config_data(config_data)

            config_path = get_config_path(config_name)
            if os.path.exists(config_path) and not overwrite:
                return JSONResponse(
                    {
                        "error": "config_exists",
                        "config": config_name,
                        "configs": get_config_names(),
                    },
                    status_code=409,
                )

            os.makedirs(os.path.dirname(config_path), exist_ok=True)
            with open(config_path, "w", encoding="utf-8", newline="\n") as f:
                json.dump(config_data, f, indent=4, ensure_ascii=False)
                f.write("\n")

            configs = get_config_names()
            return JSONResponse(
                {
                    "config": config_name,
                    "message": "overwritten" if overwrite else "uploaded",
                    "configs": configs,
                    "default": get_default_config(),
                }
            )
        except json.JSONDecodeError as exc:
            return JSONResponse(
                {"error": "invalid_json", "message": str(exc)},
                status_code=400,
            )
        except ValueError as exc:
            error_text = str(exc)
            if error_text.startswith("invalid_alas_config:"):
                parts = error_text.split(":", 2)
                return JSONResponse(
                    {
                        "error": "invalid_alas_config",
                        "reason": parts[1] if len(parts) > 1 else "",
                        "message": parts[2] if len(parts) > 2 else "",
                    },
                    status_code=400,
                )
            return JSONResponse(
                {"error": str(exc) or "invalid_config_name"},
                status_code=400,
            )
        except Exception as exc:
            return JSONResponse(
                {"error": "upload_failed", "message": str(exc)},
                status_code=500,
            )

    async def api_delete_config(request):
        config_name = get_requested_config(request)
        configs = get_config_names()
        if config_name not in configs:
            return JSONResponse(
                {
                    "error": "unknown_config",
                    "config": config_name,
                    "configs": configs,
                },
                status_code=404,
            )
        if len(configs) <= 1:
            return JSONResponse(
                {
                    "error": "cannot_delete_last_config",
                    "config": config_name,
                    "configs": configs,
                },
                status_code=409,
            )

        try:
            try:
                status = get_status(config_name)
            except Exception:
                status = "idle"
            if status == "running":
                return JSONResponse(
                    {
                        "error": "cannot_delete_running_config",
                        "config": config_name,
                        "status": "running",
                    },
                    status_code=409,
                )

            config_path = get_config_path(config_name)
            if not os.path.exists(config_path):
                return JSONResponse(
                    {
                        "error": "config_file_not_found",
                        "config": config_name,
                        "configs": configs,
                    },
                    status_code=404,
                )
            os.remove(config_path)
            configs = get_config_names()
            return JSONResponse(
                {
                    "config": config_name,
                    "message": "deleted",
                    "configs": configs,
                    "default": get_default_config(),
                }
            )
        except Exception as exc:
            return JSONResponse(
                {
                    "error": "delete_failed",
                    "config": config_name,
                    "message": str(exc),
                },
                status_code=500,
            )

    async def api_get_status_all(request):
        statuses = {}
        errors = {}
        for config_name in get_config_names():
            try:
                statuses[config_name] = get_status(config_name)
            except Exception as exc:
                statuses[config_name] = "error"
                errors[config_name] = str(exc)

        payload = {"statuses": statuses}
        if errors:
            payload["errors"] = errors
        return JSONResponse(payload)

    async def api_get_status(request):
        config_name = get_requested_config(request)
        error_response = validate_requested_config(request, config_name)
        if error_response is not None:
            return error_response

        try:
            status = get_status(config_name)
        except Exception as exc:
            return JSONResponse(
                {
                    "config": config_name,
                    "status": "error",
                    "error": str(exc),
                },
                status_code=500,
            )
        return JSONResponse({"config": config_name, "status": status})

    async def api_post_start(request):
        config_name = get_requested_config(request)
        error_response = validate_requested_config(request, config_name)
        if error_response is not None:
            return error_response

        try:
            manager = get_manager(config_name)
            already_running = bool(getattr(manager, "alive", False))
            if not already_running:
                manager.start(None)
            return JSONResponse(
                {
                    "config": config_name,
                    "message": "already_running" if already_running else "started",
                    "status": get_status(config_name),
                }
            )
        except Exception as exc:
            return JSONResponse(
                {
                    "config": config_name,
                    "message": "start_failed",
                    "status": "error",
                    "error": str(exc),
                },
                status_code=500,
            )

    async def api_post_stop(request):
        config_name = get_requested_config(request)
        error_response = validate_requested_config(request, config_name)
        if error_response is not None:
            return error_response

        try:
            manager = get_manager(config_name)
            was_running = bool(getattr(manager, "alive", False))
            if was_running:
                manager.stop()
            return JSONResponse(
                {
                    "config": config_name,
                    "message": "stopped" if was_running else "already_stopped",
                    "status": get_status(config_name),
                }
            )
        except Exception as exc:
            return JSONResponse(
                {
                    "config": config_name,
                    "message": "stop_failed",
                    "status": "error",
                    "error": str(exc),
                },
                status_code=500,
            )

    async def api_get_log(request):
        config_name = get_requested_config(request)
        error_response = validate_requested_config(request, config_name)
        if error_response is not None:
            return error_response

        line_limit = get_log_line_limit(request)
        live_error = None
        try:
            live_log = get_live_log(config_name, line_limit)
        except Exception as exc:
            live_log = ""
            live_error = str(exc)

        if live_log:
            return JSONResponse(
                {
                    "config": config_name,
                    "exists": True,
                    "source": "live",
                    "lines": line_limit,
                    "log": live_log,
                }
            )

        log_file = get_latest_log_file(config_name)
        if not log_file or not os.path.exists(log_file):
            payload = {
                "config": config_name,
                "exists": False,
                "source": "none",
                "lines": line_limit,
                "log": "",
            }
            if live_error:
                payload["live_error"] = live_error
            return JSONResponse(payload)

        try:
            payload = {
                "config": config_name,
                "exists": True,
                "source": "file",
                "file": os.path.basename(log_file),
                "lines": line_limit,
                "log": tail_log_file(log_file, line_limit),
            }
            if live_error:
                payload["live_error"] = live_error
            return JSONResponse(payload)
        except Exception as exc:
            return JSONResponse(
                {
                    "config": config_name,
                    "exists": True,
                    "source": "file",
                    "file": os.path.basename(log_file),
                    "lines": line_limit,
                    "log": "",
                    "error": str(exc),
                },
                status_code=500,
            )

    routes.append(Route("/api/{path:path}", api_options, methods=["OPTIONS"]))
    routes.append(Route("/api/health", api_health, methods=["GET"]))
    routes.append(Route("/api/configs", api_get_configs, methods=["GET"]))
    routes.append(Route("/api/configs/upload", api_upload_config, methods=["POST"]))
    routes.append(Route("/api/configs", api_delete_config, methods=["DELETE"]))
    routes.append(Route("/api/status", api_get_status, methods=["GET"]))
    routes.append(Route("/api/status_all", api_get_status_all, methods=["GET"]))
    routes.append(Route("/api/start", api_post_start, methods=["POST"]))
    routes.append(Route("/api/stop", api_post_stop, methods=["POST"]))
    routes.append(Route("/api/log", api_get_log, methods=["GET"]))

    if static_dir:
        routes.append(
            Mount("/static", app=StaticFiles(directory=static_dir), name="static")
        )
    routes.append(
        Mount(
            "/pywebio_static",
            app=StaticFiles(directory=STATIC_PATH),
            name="pywebio_static",
        )
    )
    middleware = [Middleware(HeaderMiddleware)]
    return Starlette(
        routes=routes, middleware=middleware, debug=debug, **starlette_settings
    )


def start_server(
    applications,
    port=0,
    host="",
    cdn=True,
    static_dir=None,
    remote_access=False,
    debug=False,
    allowed_origins=None,
    check_origin=None,
    auto_open_webbrowser=False,
    **uvicorn_settings
):

    app = asgi_app(
        applications,
        cdn=cdn,
        static_dir=static_dir,
        debug=debug,
        allowed_origins=allowed_origins,
        check_origin=check_origin,
    )

    if auto_open_webbrowser:
        asyncio.get_event_loop().create_task(
            open_webbrowser_on_server_started("localhost", port)
        )

    if not host:
        host = "0.0.0.0"

    if port == 0:
        port = get_free_port()

    if remote_access:
        start_remote_access_service(local_port=port)

    uvicorn.run(app, host=host, port=port, **uvicorn_settings)
